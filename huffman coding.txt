huffman coding

#include <stdio.h>

int main()
{
    int n,i,j;
    char symbol[20];
    float prob[20];

    printf("Enter number of symbols: ");
    scanf("%d",&n);

    printf("Enter symbols:\n");
    for(i=0;i<n;i++)
        scanf(" %c",&symbol[i]);

    printf("Enter probabilities:\n");
    for(i=0;i<n;i++)
        scanf("%f",&prob[i]);

    printf("\nSymbol\tProbability\n");
    for(i=0;i<n;i++)
        printf("%c\t%f\n",symbol[i],prob[i]);

    printf("\nHuffman Coding \n");

    for(i=0;i<n-1;i++)
    {
        int min1=0,min2=1;

        if(prob[min2]<prob[min1])
        {
            int temp=min1;
            min1=min2;
            min2=temp;
        }

        for(j=2;j<n-i;j++)
        {
            if(prob[j]<prob[min1])
            {
                min2=min1;
                min1=j;
            }
            else if(prob[j]<prob[min2])
                min2=j;
        }

        printf("Combine %c and %c -> %f\n",
               symbol[min1],symbol[min2],
               prob[min1]+prob[min2]);

        prob[min1]=prob[min1]+prob[min2];
        prob[min2]=prob[n-i-1];
        symbol[min2]=symbol[n-i-1];
    }

    return 0;
}
