selection sort

#include <stdio.h>
int main() {
    int i, j,k, min, n, t;
    printf("Enter the size of array:\n");
    scanf("%d", &n);
    int arr[n];
    printf("Enter %d elements:\n", n);
    for(i = 0; i < n; i++)
        scanf("%d", &arr[i]);
    printf("\nBEFORE SORTING:\n");
    for(i = 0; i < n; i++)
        printf("%d\t", arr[i]);
    printf("\n");


    for(i = 0; i < n - 1; i++) {
        min = i;
        for(j = i + 1; j < n; j++) {
            if(arr[j] < arr[min]) {
                min = j;
            }
        }

        if(min != i) {
            t = arr[min];
            arr[min] = arr[i];
            arr[i] = t;
        }

        printf("\nPass %d: ", i+1);
        for(k = 0; k < n; k++)
            printf("%d\t", arr[k]);
    }

    printf("\nAFTER SORTING:\n");
    for(i = 0; i < n; i++)
        printf("%d\t", arr[i]);
    return 0;
}

bubble sort

#include <stdio.h>

int main() {
    int i,j,k,n;
    printf("Enter the size of array: ");
    scanf("%d", &n);

    int arr[n];
    printf("Enter %d elements: ", n);
    for(i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    printf("BEFORE SORTING: ");
    for( i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    // Bubble Sort with print after each swap
    for(i = 0; i < n - 1; i++) {
        for(j = 0; j < n - i - 1; j++) {
            if(arr[j] > arr[j+1]) {
                int temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;

                printf("Pass %d: ", i+1);
                for(k = 0; k < n; k++) {
                    printf("%d ", arr[k]);
                }
                printf("\n");
            }
        }
    }

    printf("AFTER SORTING: ");
    for(i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
