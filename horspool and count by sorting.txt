horspool

#include <stdio.h>
#include <string.h>

int table[256];

void shiftTable(char pattern[])
{
    int m = strlen(pattern);
    int i;

    for(i = 0; i < 256; i++)
        table[i] = m;

    for(i = 0; i < m - 1; i++)
        table[(int)pattern[i]] = m - 1 - i;
}

int horspool(char text[], char pattern[])
{
    int n = strlen(text);
    int m = strlen(pattern);
    int i = m - 1;
    int k;

    while(i < n)
    {
        k = 0;

        while(k < m && pattern[m-1-k] == text[i-k])
            k++;

        if(k == m)
            return i - m + 1;
        else
            i = i + table[text[i]];
    }

    return -1;
}

int main()
{
    char text[100], pattern[100];
    int pos;

    printf("Enter text: ");
    scanf("%s", text);

    printf("Enter pattern: ");
    scanf("%s", pattern);

    shiftTable(pattern);

    pos = horspool(text, pattern);

    if(pos >= 0)
        printf("Pattern found at position %d\n", pos);
    else
        printf("Pattern not found\n");

    return 0;
}

#include <stdio.h>

int main() {
    int n, i, j; // Declare loop variables here

    printf("Enter number of elements: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        printf("Invalid input.\n");
        return 1;
    }
    int A[n], Count[n], S[n];

    printf("Enter elements:\n");
    for (i = 0; i < n; i++) {
        scanf("%d", &A[i]);
        Count[i] = 0;
    }


    for (i = 0; i < n - 1; i++) {
        for (j = i + 1; j < n; j++) {
            if (A[i] < A[j]) {
                Count[j]++;
            } else {
                Count[i]++;
            }
        }
    }
    for (i = 0; i < n; i++) {
        S[Count[i]] = A[i];
    }

    printf("\nSorted Array:\n");
    for (i = 0; i < n; i++) {
        printf("%d ", S[i]);
    }
    printf("\n");

    return 0;
}
