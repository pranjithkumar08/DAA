merge sort

#include<stdio.h>

void printArray(int A[], int l, int r) {
    int i;
    for(i = l; i <= r; i++) printf("%d ", A[i]);
    printf("\n");
}

void mergesort(int A[], int l, int m, int r) {

    int i = l, j = m+1, k = l;
    int B[100];
    while(i <= m && j <= r) {
        if(A[i] <= A[j]) {
            B[k++] = A[i++];
        } else {
            B[k++] = A[j++];
        }
    }

    while(i <= m) B[k++] = A[i++];
    while(j <= r) B[k++] = A[j++];

    for(i = l; i <= r; i++) {
        A[i] = B[i];
    }

    printArray(A, l, r);  // Shows result after merge
}

void merge(int A[], int l, int r) {
    if(l < r) {
        int m = l + (r - l) / 2;
        merge(A, l, m);        // Recurse left
        merge(A, m+1, r);      // Recurse right
        mergesort(A, l, m, r); // Merge them
    }
}

int main() {
    int n, i;
    printf("Enter number of elements: ");
    scanf("%d", &n);

    int A[100];
    printf("Enter %d elements: ", n);
    for(i = 0; i < n; i++) scanf("%d", &A[i]);

    printf("\n STARTING MERGE SORT\n");
    printArray(A, 0, n-1);
    merge(A, 0, n-1);
    printf("\n AFTER SORTING\n");
    printArray(A, 0, n-1);

    return 0;
}

#include<stdio.h>

void printArray(int A[], int low, int high) {
    int i;
    for(i = low; i <= high; i++) printf("%d ", A[i]);
    printf("\n");
}

int partition(int A[], int low, int high) {
    int p = A[low];  // p <- A[l]
    int i = low + 1;  // i starts at low+1
    int j = high;     // j starts at high
    while(i <= j) {
        while(i <= j && A[i] < p) i++;
        while(j >= i && A[j] > p) j--;
        if(i <= j) {
            int temp = A[i];
            A[i] = A[j];
            A[j] = temp;
            //printArray(A, low, high);
        }
    }

    int temp = A[low];
    A[low] = A[j];
    A[j] = temp;

    printf(" pos of pivot: %d\n", j);
    return j;
}

void quicksort(int A[], int l, int r) {
    if(l < r) {
        int pi = partition(A, l, r);
        quicksort(A, l, pi-1);
        quicksort(A, pi+1, r);
    }
}
