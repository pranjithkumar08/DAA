NQueen

#include <stdio.h>

#define MAX_N 20

int solutionCount = 0;

int isSafe(int board[MAX_N][MAX_N], int row, int col, int n) {
    int i, j;
    for (i = 0; i < row; i++) {
        if (board[i][col] == 1) return 0;
    }

    for (i = row, j = col; i >= 0 && j >= 0; i--, j--) {
        if (board[i][j] == 1) return 0;
    }

    for (i = row, j = col; i >= 0 && j < n; i--, j++) {
        if (board[i][j] == 1) return 0;
    }

    return 1;
}

void printSolution(int board[MAX_N][MAX_N], int vector[MAX_N], int n) {
    int i, j;
    solutionCount++;
    printf("---------------------------------\n");
    printf("Solution #%d\n", solutionCount);

    printf("Matrix:\n");
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            printf("%d ", board[i][j]);
        }
        printf("\n");
    }

    printf("Vector: [");
    for (i = 0; i < n; i++) {
        printf("%d%s", vector[i], (i == n - 1 ? "" : ", "));
    }
    printf("]\n");
}

void solveNQueensAll(int board[MAX_N][MAX_N], int row, int n, int vector[MAX_N]) {
    int col;
    if (row == n) {
        printSolution(board, vector, n);
        return;
    }

    for (col = 0; col < n; col++) {
        if (isSafe(board, row, col, n)) {
            board[row][col] = 1;
            vector[row] = col + 1;

            solveNQueensAll(board, row + 1, n, vector);

            board[row][col] = 0; // Backtrack
        }
    }
}

int main() {
    int n, i, j;
    int board[MAX_N][MAX_N];
    int vector[MAX_N];
    for (i = 0; i < MAX_N; i++) {
        vector[i] = 0;
        for (j = 0; j < MAX_N; j++) {
            board[i][j] = 0;
        }
    }

    printf("Enter N (1-%d): ", MAX_N);
    if (scanf("%d", &n) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    if (n < 1 || n > MAX_N) {
        printf("Please enter a number between 1 and %d\n", MAX_N);
    } else {
        solveNQueensAll(board, 0, n, vector);
        if (solutionCount == 0) {
            printf("No solutions found for N = %d\n", n);
        } else {
            printf("---------------------------------\n");
            printf("Total solutions found: %d\n", solutionCount);
        }
    }
    return 0;
}
